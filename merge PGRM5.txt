#include <stdio.h>

int main() {
    int a[100], b[100], c[200];
    int n, m, i, j, temp;

    printf("Enter size of first array: ");
    scanf("%d", &n);

    printf("Enter elements of first array:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter size of second array: ");
    scanf("%d", &m);

    printf("Enter elements of second array:\n");
    for (i = 0; i < m; i++)
        scanf("%d", &b[i]);

    // Merge arrays
    for (i = 0; i < n; i++)
        c[i] = a[i];

    for (i = 0; i < m; i++)
        c[n + i] = b[i];

    // Sort merged array
    for (i = 0; i < n + m - 1; i++) {
        for (j = i + 1; j < n + m; j++) {
            if (c[i] > c[j]) {
                temp = c[i];
                c[i] = c[j];
                c[j] = temp;
            }
        }
    }

    printf("Merged and sorted array:\n");
    for (i = 0; i < n + m; i++)
        printf("%d ", c[i]);

    return 0;
}
